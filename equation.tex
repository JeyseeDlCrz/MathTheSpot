\item Evaluate the limit:
$$\displaystyle \lim_{x \to -\infty}{\frac{x^2 - \sin{x}}{\sqrt{x^4 + 1}}}$$

% Choices
\begin{enumerate}
    \item $0$
    \item $1$
    \item $-1$
    \item Does not exist
\end{enumerate}

% answer: 1

% solution
\begin{enumerate}
    We know that:
    \begin{align*}
        -1 &\le \sin x \le 1 \\
        1 &\ge -\sin x \ge -1 \\
        -1 &\le -\sin x \le 1 \\
        x^2 - 1 &\le x^2 - \sin x \le x^2 + 1 \\
        \frac{x^2 - 1}{\sqrt{x^4 + 1}} &\le \frac{x^2 - \sin x}{\sqrt{x^4 + 1}} \le \frac{x^2 + 1}{\sqrt{x^4 + 1}}
    \end{align*}

    then, by evaluating the lower bound limit:
    \begin{align*}
        \lim_{x \to -\infty} \frac{x^2 - 1}{\sqrt{x^4 + 1}}
        &= \lim_{x \to -\infty} \frac{x^2\left(1 - \frac{1}{x^2}\right)}{\sqrt{x^4\left(1 + \frac{1}{x^4}\right)}} \\
        &= \lim_{x \to -\infty} \frac{x^2\left(1 - \frac{1}{x^2}\right)}{\lvert x^2 \rvert \sqrt{1 + \frac{1}{x^4}}}
    \end{align*}
    where $\lvert x^2 \rvert = x^2 \quad \forall x \in \mathbb{R}$. Substituting this in:
    \begin{align*}
        \lim_{x \to -\infty} \frac{x^2\left(1 - \frac{1}{x^2}\right)}{x^2 \sqrt{1 + \frac{1}{x^4}}}
        &= \lim_{x \to -\infty} \frac{1 - \frac{1}{x^2}}{\sqrt{1 + \frac{1}{x^4}}} \\
        &= \frac{1 - 0}{\sqrt{1 + 0}} \\
        &= \frac{1}{1} = 1
    \end{align*}

    Similarly, by evaluating the upper bound limit
    \begin{align*}
        \lim_{x \to -\infty} \frac{x^2 + 1}{\sqrt{x^4 + 1}}
        &= \lim_{x \to -\infty} \frac{x^2\left(1 + \frac{1}{x^2}\right)}{\sqrt{x^4\left(1 + \frac{1}{x^4}\right)}} \\
        &= \lim_{x \to -\infty} \frac{x^2\left(1 + \frac{1}{x^2}\right)}{\lvert x^2 \rvert \sqrt{1 + \frac{1}{x^4}}}
    \end{align*}
    where $\lvert x^2 \rvert = x^2 \quad \forall x \in \mathbb{R}$. Substituting this in:
    \begin{align*}
        \lim_{x \to -\infty} \frac{x^2\left(1 + \frac{1}{x^2}\right)}{x^2 \sqrt{1 + \frac{1}{x^4}}}
        &= \lim_{x \to -\infty} \frac{1 + \frac{1}{x^2}}{\sqrt{1 + \frac{1}{x^4}}} \\
        &= \frac{1 + 0}{\sqrt{1 + 0}} \\
        &= \frac{1}{1} = 1
    \end{align*}

    Therefore, by the Squeeze Theorem:
    \[
        \boxed{\lim_{x \to -\infty} \frac{x^2 - \sin x}{\sqrt{x^4 + 1}} = 1}
    \]
\end{enumerate}


\item Determine the value of $a$ and $b$ so that the function $f$ is continuous everywhere. $$\displaystyle f(x) = \begin{cases}
    ax - 3, & x < 1, \\[10pt]
    \frac{bx - 1}{2}, & 1 \leq x \leq 3, \\[10pt]
    \frac{x^2 - (b - 2)x + a - 4}{x - 2}, & x > 3
\end{cases}$$
% Choices
\begin{enumerate}
    \item $a = \frac{7}{4},\ b = \frac{17}{2}$
    \item $a = \frac{7}{2},\ b = \frac{17}{4}$
    \item $a = \frac{17}{2},\ b = \frac{7}{4}$
    \item $a = \frac{17}{4},\ b = \frac{7}{2}$
\end{enumerate}
% answer: 4
% solution
Let 
    \[
        h(x) = \frac{x^2 - (b-2)x + a - 4}{x - 2}, \quad x > 3 \text{ or } (3, +\infty)
    \]
    The domain is $\text{dom } h = \mathbb{R} - \{x=2\}$, but since $x=2 \notin (3, +\infty)$, $x=2$ is not a candidate for possible points of discontinuity.

    Candidates for points of discontinuity:
    \[ x = 1, \quad x = 3 \]

    At $x = 1$:
    \[
        f(1) = \frac{b(1) - 1}{2} = \frac{b - 1}{2}
    \]

    Taking limits from both sides:
    \begin{align*}
        \lim_{x \to 1^+} f(x)  &= \lim_{x \to 1^+} \frac{bx - 1}{2} 
        \qquad & \lim_{x \to 1^-} f(x) &= \lim_{x \to 1^-} (ax - 3) \\
        &= \frac{b(1) - 1}{2} 
        &&= a(1) - 3 \\
        &= \frac{b - 1}{2} 
        &&= a - 3
    \end{align*}

    Since $\lim_{x \to 1^+} f(x) = \lim_{x \to 1^-} f(x)$:
    \begin{align*}
        \frac{b - 1}{2} &= a - 3 \quad \implies \quad b - 1 = 2a - 6 \implies 2a - b - 5 = 0 \quad \text{--- (1)}
    \end{align*}

    At $x = 3$:
    \[
        f(3) = \frac{b(3) - 1}{2} = \frac{3b - 1}{2}
    \]

    Taking limits from both sides:
    \begin{align*}
        \lim_{x \to 3^+} f(x) &= \lim_{x \to 3^+} \frac{x^2 - (b-2)x + a - 4}{x - 2} \\
        &= \frac{(3)^2 - (b-2)(3) + a - 4}{3 - 2} \\
        &= \frac{9 - 3b + 6 + a - 4}{1} \\
        &= a - 3b + 11
    \end{align*}
    \begin{align*}
        \lim_{x \to 3^-} f(x) &= \lim_{x \to 3^-} \frac{bx - 1}{2} \\
        &= \frac{3b - 1}{2}
    \end{align*}

    Since $\lim_{x \to 3^+} f(x) = \lim_{x \to 3^-} f(x)$:
    \begin{align*}
        a - 3b + 11 &= \frac{3b - 1}{2} \\
        2a - 6b + 22 &= 3b - 1 \\
        2a - 9b + 23 &= 0 \quad \text{--- (2)}
    \end{align*}

    \vspace{0.5em}
    Solving the System of Linear Equations
    \[
        \begin{cases}
        2a - b - 5 = 0, & \quad (1) \\
        2a - 9b + 23 = 0, & \quad (2)
        \end{cases}
    \]


    Subtracting equation (2) from equation (1), we have:
    \begin{align*}
        (2a - b - 5) - (2a - 9b + 23) &= 0 \\
        8b - 28 &= 0 \\
        b &= \frac{28}{8} = \frac{7}{2}
    \end{align*}

    Substitute $b = \frac{7}{2}$ back into equation (1):
    \begin{align*}
        2a - b - 5 &= 0 \\
        2a - \frac{7}{2} - 5 &= 0 \\
        2a &= \frac{7}{2} + 5 \\
        2a &= \frac{17}{2} \\
        a &= \frac{17}{4}
    \end{align*}

    Therefore
    \[
        \boxed{a = \frac{17}{4}, \quad b = \frac{7}{2}}
    \]

\item Evaluate the limit: $$\displaystyle \lim_{x \to 5^-}{\frac{|x - 5|}{2x^2 - 11x + 5}}$$
% Choices
\begin{enumerate}
    \item $\frac{1}{9}$
    \item $-\frac{1}{9}$
    \item $-\frac{1}{5}$
    \item $\frac{1}{5}$
\end{enumerate}
% answer: 2
% solution
Definition of absolute value:
    \[ \left\lvert x - 5 \right\rvert = 
        \begin{cases}
            x - 5, & \quad x \geq 5 \\
            -(x - 5), & \quad x < 5
        \end{cases}
    \]

Since $x \to 5^- \implies x < 5$, Thus
    \begin{align*}
        \lim_{x \to 5^-}\frac{\left\lvert x - 5\right\rvert}{2x^2 - 11x + 5} &= \lim_{x \to 5^-}\frac{-(x - 5)}{2x^2 - 11x + 5} \\
        &= \lim_{x \to 5^-}\frac{-(x - 5)}{(x - 5)(2x - 1)} \\
        &= \lim_{x \to 5^-}\frac{-1}{2x - 1} \\
        &= -\frac{1}{2(5) - 1} \\
        &= -\frac{1}{9}
    \end{align*}

Therefore,
    \[
        \boxed{\lim_{x \to 5^-}{\frac{\left\lvert x - 5 \right\rvert}{2x^2 - 11x + 5}} = -\frac{1}{9}}
    \]

\item Determine the horizontal asymptote of the function: $$\displaystyle f(x) = \frac{x^4 - 2\cos{(2x)}}{2\sqrt{x^8 - 2} + 1}$$
% Choices
\begin{enumerate}
    \item $y = \frac{1}{2}$
    \item $y = 2$
    \item $y = 1$
    \item $y = 0$
\end{enumerate}
% answer: 1
% solution
The horizontal asymptotes of any function is:
    \[
        y = \lim_{x \to \infty} f(x) \quad \text{ and } \quad y = \lim_{x \to -\infty} f(x)
    \]
as $x \to \infty$

    \[
        y =\lim_{x \to \infty} f(x) = \lim_{x \to \infty}{\frac{x^4 - 2 \cos{2x}}{2 \sqrt{x^8 - 2} + 1}}
    \]

Using the bounds of the cosine function:
\begin{align*}
    -1 &\leq \cos{2x} \leq 1 \\
    -2 &\leq 2 \cos{2x} \leq 2 \\
    -2 &\leq -2 \cos{2x} \leq 2 \\
    x^4 - 2 &\leq x^4 - 2 \cos{2x} \leq x^4 + 2 \\
    \frac{x^4 - 2}{2 \sqrt{x^8 - 2} + 1} &\leq \frac{x^4 - 2 \cos {2x}}{2 \sqrt{x^8 - 2} + 1} \leq \frac{x^4 + 2}{2 \sqrt{x^8 - 2} + 1}
\end{align*}

Evaluating the limits of the lower and upper bounds (using limit theorem):
\[
    \lim_{x \to \infty}{\frac{x^4 - 2}{2 \sqrt{x^8 - 2} + 1}} = \frac{1}{2} \quad \text{ and } \quad \lim_{x \to \infty}\frac{x^4 - 2}{2 \sqrt{x^8 - 2} + 1} = \frac{1}{2}
\]

By Squeeze Theorem:
\[
    \lim_{x \to \infty}{\frac{x^4 - 2 \cos{2x}}{2 \sqrt{x^8 - 2} + 1}} = \frac{1}{2}
\]  

Thus, $f$ has a horizontal asymptote at $y = \frac{1}{2}$

Similarly, as $x \to -\infty$
\[
    \lim_{x \to -\infty}{\frac{x^4 - 2 \cos{2x}}{2 \sqrt{x^8 - 2} + 1}} = \frac{1}{2}
\]

Thus, $f$ has a horizontal asymptote at $y = \frac{1}{2}$

Therefore,
\[
    \boxed{y = \frac{1}{2} \text{ is the only horizontal asymptote of } f}
\]

\item Evaluate the limit: $$\displaystyle \lim_{x \to 2}\frac{x^3 + x^2 - 4x - 4}{x - 2}$$
% Choices
\begin{enumerate}
    \item $12$
    \item $16$
    \item $4$
    \item $8$
\end{enumerate}
% answer: 1
% solution
Factoring by grouping:
\begin{align*}
    x^3 + x^2 - 4x - 4 &= x^2 (x + 1) - 4 (x + 1) \\
    &= (x + 1)(x^2 - 4) \\
    &= (x + 1)(x - 2)(x + 2)
\end{align*}
Then,
\begin{align*}
    \lim_{x \to 2}{\frac{x^3 + x^2 - 4x - 4}{x - 2}} &= \lim_{x \to 2}{\frac{(x + 1)(x - 2)(x + 2)}{x - 2}} \\
    &= \lim_{x \to 2}(x + 1)(x + 2) \\
    &= (2 + 1)(2 + 2) \\
    &= 12
\end{align*}

Therfore,
\[
    \boxed{\lim_{x \to 2}{\frac{x^3 + x^2 - 4x - 4}{x - 2}} = 12}
\]
\end{enumerate}
